\section{Building a slide}

\begin{frame}{One idea, revealed one step at a time}{Progressive build}
  Duplicate the slide, add one element, keep the title. Three devices mark what
  is new, and nothing else:

  \begin{itemize}
    \item<1-> A \term{red underline} under the term being discussed
    \item<2-> A thin red box around the statement that just landed
    \item<3-> Greying out what is not yet in play
  \end{itemize}

  \vspace{1ex}
  \onslide<2->{\keybox[0.6\linewidth]{Aligned $m \to m+1$, anti-aligned $m \to m-1$}}

  \vspace{1ex}
  \reveal<3>{This line is greyed out until the third overlay.}
\end{frame}

\begin{frame}{The equation is the anchor}{Master equation}
  A content slide carries one anchor --- an equation, a figure or a schematic ---
  never two.

  \begin{equation*}
    \partial_t \rho_m
      = -i\left[\term{\bm{B}_m\cdot\bm{S}},\rho_m\right]
        + \Gamma_m^{+}\mathcal{D}[\sigma^{\dagger}]\rho_m
        + \Gamma_m^{-}\mathcal{D}[\sigma]\rho_m
  \end{equation*}

  \begin{itemize}
    \item $\rho_S = \sum_m \rho_m$, with $\mathrm{tr}\,\rho_m = P_m$
    \item Symbols are defined before they are used
  \end{itemize}

  \slidecite{A.~Riera-Campeny, A.~Sanpera, P.~Strasberg, \emph{Quantum systems
    correlated with a finite bath}, PRX Quantum \textbf{2}, 010340 (2021)}
\end{frame}
